cmake_minimum_required(VERSION 3.22)
project(FET67 VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FET67_COPY_PLUGIN_AFTER_BUILD "Install VST3 after building (may require elevation)" ON)
set(FET67_JUCE_PATH "C:/JUCE" CACHE PATH "Optional JUCE 7/8/9 source checkout")
if(EXISTS "${FET67_JUCE_PATH}/CMakeLists.txt")
    file(STRINGS "${FET67_JUCE_PATH}/CMakeLists.txt" _juce_version REGEX "project\\(JUCE VERSION [789]\\.")
endif()
if(_juce_version)
    add_subdirectory("${FET67_JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
else()
    include(FetchContent)
    FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/9.0.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(FET67
    COMPANY_NAME "Papielguapo"
    BUNDLE_ID "com.papielguapo.fet67"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD ${FET67_COPY_PLUGIN_AFTER_BUILD}
    PLUGIN_MANUFACTURER_CODE Papi PLUGIN_CODE Ft67
    FORMATS VST3 Standalone PRODUCT_NAME "FET 67")
target_sources(FET67 PRIVATE Source/PluginProcessor.h Source/PluginProcessor.cpp Source/PluginEditor.h Source/PluginEditor.cpp)
target_compile_definitions(FET67 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(FET67 PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
option(FET67_BUILD_TESTS "Build offline DSP and reference measurement tool" OFF)
if(FET67_BUILD_TESTS)
    enable_testing()
    juce_add_console_app(FET67Tests PRODUCT_NAME "FET67Tests")
    target_sources(FET67Tests PRIVATE Tests/Tests.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
    target_compile_definitions(FET67Tests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(FET67Tests PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
    add_test(NAME DSP COMMAND FET67Tests)
    if(WIN32)
        add_executable(FET67ReferenceCatalog Tests/ReferenceCatalog.cpp)
        if(EXISTS "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
            target_include_directories(FET67ReferenceCatalog PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
        else()
            target_include_directories(FET67ReferenceCatalog PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors/format_types/VST3_SDK")
        endif()
    endif()
endif()
